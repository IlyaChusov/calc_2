Option Explicit

Sub Рассчитать()
    
    speedUpMacros True
    
    On Error GoTo ErrorExit
    '''''''''''''''
    Dim sheetBrief As Worksheet, allBrands As New Collection, temp As Range, temp2 As Range, tmpStr As String, tempStr As String, plusBrandCell As Range, i As Integer
    Set sheetBrief = ActiveWorkbook.Sheets("Бриф")
    'Set temp = sheetBrief.UsedRange.Find("+ Бренд")
    If temp Is Nothing Then Call Err.Raise(10, , "Не найдены бренды для добавления на листе Бриф")
    tmpStr = temp.Address
    allBrands.Add temp
    Set temp = sheetBrief.UsedRange.FindNext
    Do While True
        Set temp = sheetBrief.UsedRange.FindNext(temp)
        If temp.Address = tmpStr Then Exit Do
        allBrands.Add temp
    Loop
    
    Dim startCellBook As Range, caBook As Object, brandBookName As String
    Set caBook = CreateObject("Scripting.Dictionary")
    Set startCellBook = sheetBrief.Cells(9, 5)
    If startCellBook.Value <> "Бренд" Then Call Err.Raise(10, , "Не найдена область со справочником ЦА-Бренд")
    Set startCellBook = startCellBook.Offset(1, 0)
    For i = 0 To 12
        brandBookName = Trim(startCellBook.Value)
        If brandBookName <> "|" And brandBookName <> "" Then
            caBook.Item(brandBookName) = startCellBook.Offset(0, -2).Address(False, False)
        End If
        Set startCellBook = startCellBook.Offset(1, 0)
    Next i
    
    '''''''''''''''
    Dim sheetTech As Worksheet, regionCellTech As Range
    Set sheetTech = ActiveWorkbook.Sheets("тех")
    Set regionCellTech = sheetTech.UsedRange.Find("Регион")
    If regionCellTech Is Nothing Then Call Err.Raise(10, , "Не найдена стартовая ячейка с регионами на листе тех")
    
    '''''''''''''''
    Dim sheetAff As Worksheet, startCellAff As Range
    Set sheetAff = ActiveWorkbook.Sheets("Aff")
    Set startCellAff = sheetAff.UsedRange.Find("город")
    If startCellAff Is Nothing Then Call Err.Raise(10, , "Не найдена стартовая ячейка с городом на листе Aff")
    Set startCellAff = startCellAff.Offset(1, 0)
    Do While startCellAff <> ""
        Set startCellAff = startCellAff.Offset(1, 0)
    Loop
    
    '''''''''''''''
    Dim sheetReach As Worksheet, startCellReach As Range
    Set sheetReach = ActiveWorkbook.Sheets("Reach")
    Set startCellReach = sheetReach.UsedRange.Find("город")
    If startCellReach Is Nothing Then Call Err.Raise(10, , "Не найдена стартовая ячейка с городом на листе Reach")
    Set startCellReach = startCellReach.Offset(1, 0)
    Do While startCellReach <> ""
        Set startCellReach = startCellReach.Offset(1, 0)
    Loop
    
    '''''''''''''''
    Dim sheetMP As Worksheet, startCellMP As Range, lastCellMP As Range, totalTVRColMP As Integer, canalKoefColMP As Integer, clientKoefColMP As Integer, priceBaseColMP As Integer, avTVRColMP As Integer, affColMP As Integer, splitColMP As Integer
    Set sheetMP = ActiveWorkbook.Sheets("МП")
    Set startCellMP = sheetMP.UsedRange.Find("Город")
    totalTVRColMP = sheetMP.UsedRange.Find("total TVR ЦА").Column
    Set temp = sheetMP.UsedRange.Find("Split")
    If temp Is Nothing Then Call Err.Raise(10, , "Не найдена колонка Split на листе МП")
    splitColMP = temp.Column
    Set temp = sheetMP.UsedRange.Find("Бонус закладываем").Offset(0, -1)
    canalKoefColMP = temp.Column
    clientKoefColMP = temp.Offset(0, -1).Column
    priceBaseColMP = sheetMP.UsedRange.Find("Цена базовая", lookat:=xlPart).Column
    avTVRColMP = sheetMP.UsedRange.Find("av TVR БА").Column
    affColMP = sheetMP.UsedRange.Find("Aff").Column
    If startCellMP Is Nothing Then Call Err.Raise(10, , "Не найдена стартовая ячейка на листе МП")
    Set startCellMP = startCellMP.Offset(2, 0)
    Set lastCellMP = startCellMP
    Do While lastCellMP <> ""
        Set lastCellMP = lastCellMP.Offset(1, 0)
    Loop
    Set lastCellMP = lastCellMP.Offset(-1, 0)
    
    '''''''''''''''
    Dim sheetFlow As Worksheet, startRowFlow As Integer, tcppCell As Range, tcppFirstAdr As String, tcppPrevCol As Integer
    Set sheetFlow = ActiveWorkbook.Sheets("FLOW")
    i = 0
    For Each temp In sheetFlow.Range("D:D")
        If i > 2000 Then Exit For
        If temp = "Budget total" Then
            startRowFlow = temp.Row + 1
            Exit For
        End If
        i = i + 1
    Next temp
    
    '''''''''''''''
    Dim sheetSdelka As Worksheet, startCellSdelka As Range, lastCellSdelka As Range, newLastCellSdelka As Range, ccColSdelka As Integer, trpNacColSdelka As Integer, splitCaColSdelka As Integer, chronColSdelka As Integer, chronCoefColSdelka As Integer, affColSdelka As Integer
    Set sheetSdelka = ActiveWorkbook.Sheets("СДЕЛКА")
    Set startCellSdelka = sheetSdelka.UsedRange.Find("Город")
    ccColSdelka = sheetSdelka.UsedRange.Find("Grp20 с наценкой на хрон").Offset(0, 1).Column
    trpNacColSdelka = sheetSdelka.UsedRange.Find("Trp с наценкой на хрон").Column
    splitCaColSdelka = sheetSdelka.UsedRange.Find("Сплит ЦА").Column
    chronColSdelka = sheetSdelka.UsedRange.Find("Хрон", lookat:=xlWhole).Column
    chronCoefColSdelka = sheetSdelka.UsedRange.Find("Коэф. на хрон", lookat:=xlWhole).Column
    affColSdelka = sheetSdelka.UsedRange.Find("Aff").Column
    If startCellSdelka Is Nothing Then Call Err.Raise(10, , "Не найдена стартовая ячейка на листе СДЕЛКА")
    Set startCellSdelka = startCellSdelka.Offset(2, 0)
    Set lastCellSdelka = startCellSdelka
    Do While lastCellSdelka <> ""
        Set lastCellSdelka = lastCellSdelka.Offset(1, 0)
    Loop
    Set lastCellSdelka = lastCellSdelka.Offset(-1, 0)
    
    'Лист, с которого будут копироваться новые блоки
    Dim sheetCopy As Worksheet
    Set sheetCopy = ActiveWorkbook.Sheets("Copy")
    
    
    
    
    
    Dim brandName As String, caName As String, caNum As String, regions_ As Range, regions As Collection, regionName As Variant, caAndBrand As String
    Dim regionTargetTech As Range, regionTargetLastTech As Range, regionRowsTech As Integer
    Dim newLastCellMP As Range, oldLastCellMP As Range, arr As Variant
'    Dim allBrandsOnFlow
'    Set allBrandsOnFlow = CreateObject("Scripting.Dictionary")
'    For Each temp In sheetFlow.Range("C:C")
'        If i > 2000 Then Exit For
'        If temp.Interior.ColorIndex = 6 Then
'            arr = Split(temp.Value, "_")
'            brandName = arr(0)
'            caName = ""
'            If UBound(arr) > 1 Then caName = arr(1)
'            caNum = temp.Offset(0, -1).Value
'            allBrandsOnFlow
'        End If
'        i = i + 1
'    Next temp
    
    
    'Ячейка по столбцу av TVR ЦА, с которой ошибки. Поэтому ее для второго и далее регионов копирую с первого, а записываю в эту переменную
    'TODO: подумать над коллекцией всех ячеек с ошибкой вместо отдельной переменной для каждой
    Dim savedCellReach As Range, savedCellavTVRCA As Range, savedCellTotalTVR As Range, savedCellCanalCoef As Range, savedCellClientCoef As Range, savedCellPriceBase As Range, savedCellavTVRBA As Range, savedCellAff As Range
    Dim savedCellChronSdelka As Range, savedCellChronCoefSdelka As Range, savedCellTrpSumSdelka As Range
    Dim cellsWithErrorCollection As Collection
    Dim check, caCell As Range, brandCell As Range, regionCell As Range, caNumAdr As String, regionAdr As String, chronsNot As New Collection, j As Integer, rowChronFlowNum As Integer, t As Variant
    For Each check In sheetBrief.CheckBoxes
        If check.Value = -4146 Then chronsNot.Add check.Caption
    Next check
    For Each plusBrandCell In allBrands
        Set brandCell = plusBrandCell.Offset(0, 2)
        brandName = brandCell.Value
        With plusBrandCell.Offset(1, 2)
            caNumAdr = "=" & sheetBrief.Name & "!" & .Address(False, False)
            caNum = Trim(.Value)
        End With
        sheetFlow.Cells(2, 2) = sheetFlow.Cells(2, 2) & "," & brandName
        caName = ""
        Set caCell = Nothing
        If caBook.exists(brandName) Then
            Set caCell = sheetBrief.Range(caBook(brandName))
            caName = Trim(caCell)
            caAndBrand = "=CONCAT(" & sheetBrief.Name & "!" & brandCell.Address(False, False) & ", ""_"", " & sheetBrief.Name & "!" & caCell.Address(False, False) & ")"
        Else
            caAndBrand = "=CONCAT(" & sheetBrief.Name & "!" & brandCell.Address(False, False) & ", ""_"")"
        End If
        Set regions_ = sheetBrief.Range(plusBrandCell.Offset(4, 0), plusBrandCell.Offset(44, 0))
        Set regions = New Collection
        For Each temp In regions_
            If temp <> "" Then regions.Add temp
        Next temp
        
        For Each regionCell In regions
            regionName = Trim(regionCell.Value)
            regionAdr = "=" & sheetBrief.Name & "!" & regionCell.Address(False, False)
            
            
            
            
            'ЛИСТ ФЛОУ
            If Trim(regionName) = "Регионы" Then regionName = ""
            For i = 0 To 20
                sheetFlow.Rows(startRowFlow - 1).EntireRow.Insert
            Next i
            sheetCopy.Range("A60:BX80").Copy (sheetFlow.Rows(startRowFlow - 1))
            sheetFlow.Range(sheetFlow.Cells(startRowFlow - 1, 1), sheetFlow.Cells(startRowFlow + 18, 1)).EntireRow.Group
            sheetFlow.Range(sheetFlow.Cells(startRowFlow + 7, 1), sheetFlow.Cells(startRowFlow + 15, 1)).EntireRow.Group
            sheetFlow.Cells(startRowFlow + 1, 2).Formula = caNumAdr
            sheetFlow.Cells(startRowFlow + 1, 3).Formula = caAndBrand
            sheetFlow.Cells(startRowFlow + 1, 1).Formula = regionAdr
            'Удаление неотмеченных строк Хрона на листе Флоу
            j = 0
            For Each check In chronsNot
                If check = "25" Then
                    rowChronFlowNum = 11
                ElseIf check = "20" Then
                    rowChronFlowNum = 12
                ElseIf check = "15" Then
                    rowChronFlowNum = 13
                ElseIf check = "10" Then
                    rowChronFlowNum = 14
                ElseIf check = "5" Then
                    rowChronFlowNum = 15
                End If
                sheetFlow.Cells(startRowFlow + rowChronFlowNum - j, 4).EntireRow.Delete
                j = j + 1
            Next check
            startRowFlow = startRowFlow + 21 - j
            
            Set regionTargetTech = regionCellTech.EntireColumn.Find(regionName)
            If regionTargetTech Is Nothing Then Call Err.Raise(10, , "Не найден нужный регион на листе тех")
            Set regionTargetLastTech = regionTargetTech
            Do While regionTargetLastTech.Value = regionName
                Set regionTargetLastTech = regionTargetLastTech.Offset(1, 0)
            Loop
            Set regionTargetLastTech = regionTargetLastTech.Offset(-1, 0)
            regionRowsTech = regionTargetLastTech.Row - regionTargetTech.Row
            
            
            
            
            
            'ЛИСТ МП (и немного Сделка)
            For i = 0 To regionRowsTech - 1
                sheetCopy.Rows(96).EntireRow.Copy sheetMP.Cells(lastCellMP.Row + 1 + i, 1)
                sheetCopy.Rows(86).EntireRow.Copy sheetSdelka.Cells(lastCellSdelka.Row + 1 + i, 1)
            Next i
            sheetCopy.Rows(97).EntireRow.Copy sheetMP.Cells(lastCellMP.Row + 1 + i, 1)
            sheetCopy.Rows(87).EntireRow.Copy sheetSdelka.Cells(lastCellSdelka.Row + 1 + i, 1)
            
            sheetTech.Range(regionTargetTech, regionTargetLastTech.Offset(0, 3)).Copy (sheetMP.Cells(lastCellMP.Row + 1, 1))
            sheetCopy.Cells(97, 4).Copy sheetMP.Cells(lastCellMP.Row + 1 + i, 4)
            Set newLastCellMP = sheetMP.Cells(lastCellMP.Row + regionRowsTech + 1, 1)
            
            Dim regionMPCell As Range
            Set regionMPCell = sheetMP.Cells(lastCellMP.Row + 1, 1)
            With sheetMP.Cells(lastCellMP.Row + 1, 1)
                .Formula = regionAdr
                sheetMP.Range(.Offset(1, 0), newLastCellMP).Formula = "=" & .Address(True, False)
            End With

            Dim caNumCell As Range
            Set caNumCell = sheetMP.Cells(lastCellMP.Row + 1, 5)
            With sheetMP.Cells(lastCellMP.Row + 1, 5)
                .Formula = caNumAdr
                sheetMP.Range(.Offset(1, 0), sheetMP.Cells(lastCellMP.Row + regionRowsTech + 1, 5)).Formula = "=" & .Address(True, False)
            End With
            
            Dim caAndBrandCell As Range
            Set caAndBrandCell = sheetMP.Cells(lastCellMP.Row + 1, 6)
            With sheetMP.Cells(lastCellMP.Row + 1, 6)
                .Formula = caAndBrand
                sheetMP.Range(.Offset(1, 0), sheetMP.Cells(lastCellMP.Row + regionRowsTech + 1, 6)).Formula = "=" & .Address(True, False)
            End With
            
            sheetMP.Cells(newLastCellMP.Row, 6).Formula = caAndBrand
            sheetMP.Cells(lastCellMP.Row + 1, 4).Copy
            sheetMP.Range(sheetMP.Cells(lastCellMP.Row + 1, 5), newLastCellMP.Offset(-1, 5)).PasteSpecial xlPasteFormats
            
            With sheetMP.Cells(lastCellMP.Row + 1, splitColMP)
                .Formula = "=" & .Offset(0, 1).Address(False, False) & "/" & sheetMP.Cells(newLastCellMP.Row, splitColMP + 1).Address
                .AutoFill sheetMP.Range(sheetMP.Cells(lastCellMP.Row + 1, splitColMP), sheetMP.Cells(newLastCellMP.Row - 1, splitColMP))
            End With
            
            'Правка формул колонки Reach
            Set temp = sheetMP.Cells(lastCellMP.Row + 1, 9)
            If savedCellReach Is Nothing Then
                Set cellsWithErrorCollection = New Collection
                cellsWithErrorCollection.Add sheetMP.Cells(temp.Row, 5).Address(False)
                cellsWithErrorCollection.Add sheetMP.Cells(temp.Row, 1).Address(False)
                cellsWithErrorCollection.Add sheetMP.Cells(temp.Row, 3).Address(False, False)
                tempStr = temp.Formula
                i = 1
                j = 0
                Do While InStr(tempStr, "#REF") <> 0
                    tempStr = Replace(tempStr, "#REF!", cellsWithErrorCollection(i), 1, 1)
                    i = i + 1
                    If i > cellsWithErrorCollection.Count Then
                        i = 1
                        j = j + 1
                        If j = 7 Then
                            Dim gg As New Collection
                            For Each t In cellsWithErrorCollection
                                gg.Add t
                            Next t
                            Set cellsWithErrorCollection = New Collection
                            cellsWithErrorCollection.Add gg(2)
                            cellsWithErrorCollection.Add gg(3)
                            cellsWithErrorCollection.Add gg(1)
                            Set gg = Nothing
                            j = 8
                        End If
                    End If
                Loop
                temp.Formula = tempStr
                Set savedCellReach = temp
            Else
                savedCellReach.Copy temp
            End If
            temp.AutoFill sheetMP.Range(temp, newLastCellMP.Offset(-1, 8))
            
            'Правка формул колонки av TVR ЦА
            With sheetMP.Cells(lastCellMP.Row + 1, 10)
                Set cellsWithErrorCollection = New Collection
                cellsWithErrorCollection.Add sheetMP.Cells(.Row, 5).Address(False)
                cellsWithErrorCollection.Add sheetMP.Cells(.Row, 1).Address(False)
                cellsWithErrorCollection.Add sheetMP.Cells(.Row, 3).Address(False, False)
            End With
            Call correctBrokeFormula(sheetMP, sheetMP.Cells(lastCellMP.Row + 1, 10), savedCellavTVRCA, sheetMP.Range(sheetMP.Cells(lastCellMP.Row + 1, 10), newLastCellMP.Offset(-1, 9)), cellsWithErrorCollection)
            
            'Правка формул колонки total TVR ЦА
            With sheetMP.Cells(lastCellMP.Row + 1, totalTVRColMP)
                Set cellsWithErrorCollection = New Collection
                cellsWithErrorCollection.Add sheetMP.Cells(.Row, 18).Address(False)
                cellsWithErrorCollection.Add sheetMP.Cells(.Row, 5).Address(False, False)
                cellsWithErrorCollection.Add sheetMP.Cells(.Row, 1).Address(False)
                cellsWithErrorCollection.Add sheetMP.Cells(.Row, 3).Address(False)
            End With
            Call correctBrokeFormula(sheetMP, sheetMP.Cells(lastCellMP.Row + 1, totalTVRColMP), savedCellTotalTVR, sheetMP.Range(sheetMP.Cells(lastCellMP.Row + 1, totalTVRColMP), sheetMP.Cells(newLastCellMP.Row - 1, totalTVRColMP)), cellsWithErrorCollection)
            
            'Правка формул колонки Канал коэфф
            With sheetMP.Cells(lastCellMP.Row + 1, canalKoefColMP)
                Set cellsWithErrorCollection = New Collection
                cellsWithErrorCollection.Add sheetMP.Cells(.Row, 1).Address(False)
                cellsWithErrorCollection.Add sheetMP.Cells(.Row, 3).Address(False)
            End With
            Call correctBrokeFormula(sheetMP, sheetMP.Cells(lastCellMP.Row + 1, canalKoefColMP), savedCellCanalCoef, sheetMP.Range(sheetMP.Cells(lastCellMP.Row + 1, canalKoefColMP), sheetMP.Cells(newLastCellMP.Row - 1, canalKoefColMP)), cellsWithErrorCollection)
            
            'Правка формул колонки Клиент коэфф
            With sheetMP.Cells(lastCellMP.Row + 1, clientKoefColMP)
                Set cellsWithErrorCollection = New Collection
                cellsWithErrorCollection.Add sheetMP.Cells(.Row, 1).Address(False)
                cellsWithErrorCollection.Add sheetMP.Cells(.Row, 3).Address(False)
            End With
            Call correctBrokeFormula(sheetMP, sheetMP.Cells(lastCellMP.Row + 1, clientKoefColMP), savedCellClientCoef, sheetMP.Range(sheetMP.Cells(lastCellMP.Row + 1, clientKoefColMP), sheetMP.Cells(newLastCellMP.Row - 1, clientKoefColMP)), cellsWithErrorCollection)
            
            'Правка формул колонки Цена базовая *год*
            With sheetMP.Cells(lastCellMP.Row + 1, priceBaseColMP)
                Set cellsWithErrorCollection = New Collection
                cellsWithErrorCollection.Add sheetMP.Cells(.Row, 3).Address(False)
                cellsWithErrorCollection.Add sheetMP.Cells(.Row, 1).Address(False)
            End With
            Call correctBrokeFormula(sheetMP, sheetMP.Cells(lastCellMP.Row + 1, priceBaseColMP), savedCellPriceBase, sheetMP.Range(sheetMP.Cells(lastCellMP.Row + 1, priceBaseColMP), sheetMP.Cells(newLastCellMP.Row - 1, priceBaseColMP)), cellsWithErrorCollection)
            
            'Правка формул колонки av TVR БА
            With sheetMP.Cells(lastCellMP.Row + 1, avTVRColMP)
                Set cellsWithErrorCollection = New Collection
                cellsWithErrorCollection.Add sheetMP.Cells(.Row, 1).Address(False)
                cellsWithErrorCollection.Add sheetMP.Cells(.Row, 3).Address(False)
            End With
            Call correctBrokeFormula(sheetMP, sheetMP.Cells(lastCellMP.Row + 1, avTVRColMP), savedCellavTVRBA, sheetMP.Range(sheetMP.Cells(lastCellMP.Row + 1, avTVRColMP), sheetMP.Cells(newLastCellMP.Row - 1, avTVRColMP)), cellsWithErrorCollection)
            
            'Правка формул колонки Aff
            With sheetMP.Cells(lastCellMP.Row + 1, affColMP)
                Set cellsWithErrorCollection = New Collection
                cellsWithErrorCollection.Add sheetMP.Cells(.Row, 1).Address(False)
                cellsWithErrorCollection.Add sheetMP.Cells(.Row, 3).Address(False)
                cellsWithErrorCollection.Add sheetMP.Cells(.Row, 5).Address(False)
            End With
            Call correctBrokeFormula(sheetMP, sheetMP.Cells(lastCellMP.Row + 1, affColMP), savedCellAff, sheetMP.Range(sheetMP.Cells(lastCellMP.Row + 1, affColMP), sheetMP.Cells(newLastCellMP.Row - 1, affColMP)), cellsWithErrorCollection)

                If InStr(temp.Formula, "=SUM(") <> 0 Then
                    temp.Formula = "=SUM(" & sheetMP.Cells(lastCellMP.Row + 1, temp.Column).Address(False, False) & ":" & sheetMP.Cells(newLastCellMP.Row - 1, temp.Column).Address(False, False) & ")"
                End If
            Next temp
            
            Set oldLastCellMP = lastCellMP
            Set lastCellMP = newLastCellMP
            
            Set temp = sheetMP.Range(sheetMP.Cells(oldLastCellMP.Row + 1, 1), sheetMP.Cells(newLastCellMP.Row, 6))
            With temp
                .Copy startCellAff
                .Copy startCellReach
                
                startCellAff.Formula = "=" & sheetMP.Name & "!" & regionMPCell.Address(False, False)
                startCellReach.Formula = "=" & sheetMP.Name & "!" & regionMPCell.Address(False, False)
                
                startCellAff.Offset(0, 4).Formula = "=" & sheetMP.Name & "!" & caNumCell.Address(False, False)
                startCellReach.Offset(0, 4).Formula = "=" & sheetMP.Name & "!" & caNumCell.Address(False, False)
                
                startCellAff.Offset(0, 5).Formula = "=" & sheetMP.Name & "!" & caAndBrandCell.Address(False, False)
                startCellReach.Offset(0, 5).Formula = "=" & sheetMP.Name & "!" & caAndBrandCell.Address(False, False)
                
                For Each t In Array(0, 4, 5)

                    t = CInt(t)
                    For Each temp2 In sheetAff.Range(startCellAff.Offset(0, t), startCellAff.Offset(.Rows.Count, t))
                        If i = 0 Then i = i + 1 Else temp2.Formula = "=" & startCellAff.Offset(0, t).Address(True, False)
                    Next temp2
                    i = 0
                    For Each temp2 In sheetReach.Range(startCellReach.Offset(0, t), startCellReach.Offset(.Rows.Count, t))
                        If i = 0 Then i = i + 1 Else temp2.Formula = "=" & startCellReach.Offset(0, t).Address(True, False)
                    Next temp2
                Next t
                Set startCellAff = startCellAff.Offset(.Rows.Count, 0)
                Set startCellReach = startCellReach.Offset(.Rows.Count, 0)
            End With
            
            
            
            
            'ЛИСТ СДЕЛКА
            sheetTech.Range(regionTargetTech, regionTargetLastTech.Offset(0, 3)).Copy (sheetSdelka.Cells(lastCellSdelka.Row + 1, 1))
            Set newLastCellSdelka = sheetSdelka.Cells(lastCellSdelka.Row + regionRowsTech + 1, 1)
            
            With sheetSdelka.Cells(lastCellSdelka.Row + 1, 1)
                .Formula = regionAdr
                sheetSdelka.Range(.Offset(1, 0), newLastCellSdelka).Formula = "=" & .Address(True, False)
            End With
            sheetCopy.Cells(87, 4).Copy sheetSdelka.Cells(newLastCellSdelka.Row, 4)
            
            With sheetSdelka.Cells(lastCellSdelka.Row + 1, 5)
                .Formula = caNumAdr
                sheetSdelka.Range(.Offset(1, 0), sheetSdelka.Cells(lastCellSdelka.Row + regionRowsTech + 1, 5)).Formula = "=" & .Address(True, False)
            End With

                .Formula = caAndBrand
                sheetSdelka.Range(.Offset(1, 0), sheetSdelka.Cells(lastCellSdelka.Row + regionRowsTech + 1, 6)).Formula = "=" & .Address(True, False)
            End With
            
            sheetSdelka.Cells(lastCellSdelka.Row + 1, 4).Copy
            sheetSdelka.Range(sheetSdelka.Cells(lastCellSdelka.Row + 1, 5), newLastCellSdelka.Offset(-1, 5)).PasteSpecial xlPasteFormats
            
            Dim itogoSdelkaCol As Integer
            Set temp = sheetSdelka.Cells(1, sheetSdelka.UsedRange.Columns.Count)
            Do While temp = ""
                Set temp = temp.Offset(0, -1)
            Loop
            itogoSdelkaCol = temp.Column
            
            With sheetSdelka.Cells(lastCellSdelka.Row + 1, 7)
                .Formula = "=" & sheetMP.Name & "!" & sheetMP.Cells(oldLastCellMP.Row, 8).Address(False, False)
                .AutoFill sheetSdelka.Range(sheetSdelka.Cells(lastCellSdelka.Row + 1, 7), sheetSdelka.Cells(newLastCellSdelka.Row - 1, 7))
            End With
            
            j = 0
            For i = lastCellSdelka.Row + 1 To newLastCellSdelka.Row - 1
                With sheetSdelka.Cells(i, 9)
                    .Formula = "=" & sheetMP.Name & "!" & sheetMP.Cells(oldLastCellMP.Row + j, 18).Address(False)
                    .AutoFill sheetSdelka.Range(sheetSdelka.Cells(i, 9), sheetSdelka.Cells(i, 20))
                End With
                j = j + 1
            Next i
            
            i = 1
            'Правка формул колонок CC-CN
            For Each temp In sheetSdelka.Range(sheetSdelka.Cells(lastCellSdelka.Row + 1, ccColSdelka), sheetSdelka.Cells(newLastCellSdelka.Row - 1, ccColSdelka + 11))
                Set cellsWithErrorCollection = New Collection
                cellsWithErrorCollection.Add sheetSdelka.Cells(temp.Row, 1).Address(False)
                cellsWithErrorCollection.Add sheetSdelka.Cells(temp.Row, 3).Address(False)
                cellsWithErrorCollection.Add sheetSdelka.Cells(temp.Row, 5).Address(False)
                tempStr = temp.Formula

                Do While InStr(tempStr, "#REF") <> 0
                    tempStr = Replace(tempStr, "#REF!", cellsWithErrorCollection(i), 1, 1)
                    i = i + 1
                    If i > cellsWithErrorCollection.Count Then i = 1

Sub correctBrokeFormula(sheet As Worksheet, ByRef cell As Range, ByRef cellToCopy As Range, rangeToFill As Range, coll As Collection)
    Dim tempStr As String, i As Integer
    i = 1
    If cellToCopy Is Nothing Then
        With cell
            tempStr = .Formula
            
            Do While InStr(tempStr, "#REF") <> 0
                tempStr = Replace(tempStr, "#REF!", coll(i), 1, 1)
                i = i + 1
                If i > coll.Count Then i = 1
            Loop
            .Formula = tempStr
        End With
        Set cellToCopy = cell
    Else
        cellToCopy.Copy cell
    End If
    cell.AutoFill rangeToFill

End Sub

Sub speedUpMacros(yes As Boolean)
    Dim temp As Worksheet
    With Application
        If yes Then
            .Calculation = xlManual
            .ScreenUpdating = False
            .EnableEvents = False
        Else
            .Calculate
            .Calculation = xlAutomatic
            .ScreenUpdating = True
            .EnableEvents = True
        End If
    End With
    
    For Each temp In ActiveWorkbook.Worksheets
        temp.EnableCalculation = Not yes
        temp.EnableFormatConditionsCalculation = Not yes
    Next temp
End Sub
